# 资源页升级知识库 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 将现有资源页升级为项目知识库 MVP，支持资源入库、文本解析、摘要/标签、FTS 检索、Stitch 三栏布局和会话引用知识上下文。

**Architecture:** 复用现有 `files` 与 `resource_assets` 作为原始资源来源，在其上新增 `knowledge_sources`、`knowledge_extractions`、`knowledge_chunks` 和 FTS 索引层。上传文件与智能体文档归档后同步创建知识源，文本类资源立即解析并索引，前端资源页按 Stitch `深海指挥中心 - 知识库项目筛选版` 重排为知识库工作台。

**Tech Stack:** Node.js 20、TypeScript、Express、better-sqlite3、SQLite FTS5、React 18、Vite、TanStack Query、Tailwind/CSS tokens、node:test。

---

## 0. 范围与执行约束

本计划只覆盖 Phase 1 MVP：

- 做：数据表、repo、文本/Markdown/JSON/CSV 基础解析、图片 metadata、agent document 入库、FTS 检索、API、知识库页面布局、会话引用优先使用知识库片段。
- 不做：真实 PDF/Office/OCR 解析、向量数据库、embedding、rerank、GraphRAG、ACP 主动知识工具、跨项目共享知识。
- 不改：现有 `uploaded_file` / `agent_document` 兼容语义，不迁移旧主键，不删除现有 `files` / `resource_assets` API。

```ts
import { nanoid } from 'nanoid';
import { db, now } from '../db.js';
import type {
  KnowledgeChunk,
  KnowledgeChunkType,
  KnowledgeExtraction,
  KnowledgeSearchResult,
  KnowledgeSource,
  KnowledgeSourceListItem,
  KnowledgeSourceType,
  KnowledgeStatus,
} from '../knowledge-types.js';

type JsonRecord = Record<string, unknown>;

interface EnsureSourceInput {

  room_id?: string | null;
  source_type: KnowledgeSourceType;
  source_id: string;
  title: string;

interface SaveExtractionInput {

  markdown?: string | null;

interface ReplaceChunkInput {

  room_id?: string | null;
  chunk_index: number;
  chunk_type: KnowledgeChunkType;
  title: string;
  content: string;
  summary?: string | null;
  metadata?: JsonRecord | null;
  page_start?: number | null;
  page_end?: number | null;

interface SourceFilters {
  projectId: string;
  roomId?: string;
  statuses?: KnowledgeStatus[];
  sourceTypes?: KnowledgeSourceType[];
  query?: string;
  limit?: number;

interface SearchInput {

export const knowledgeRepo = {
  ensureSource(input: EnsureSourceInput): KnowledgeSource {
    const timestamp = now();
    const existing = db.prepare(
      `SELECT * FROM knowledge_sources
       WHERE project_id = ? AND source_type = ? AND source_id = ?`,
    ).get(input.project_id, input.source_type, input.source_id) as KnowledgeSource | undefined;

    if (existing) {
      db.prepare(
        `UPDATE knowledge_sources
         SET room_id = ?, title = ?, mime_type = ?, size = ?, status = ?, content_hash = ?,
             parser = ?, parser_version = ?, summary = ?, tags_json = ?, metadata_json = ?,
             updated_at = ?, deleted_at = NULL
         WHERE id = ?`,
      ).run(
        input.room_id ?? null,
        input.title,
        input.mime_type ?? null,
        input.size ?? null,
        input.status,
        input.content_hash ?? null,
        input.parser ?? null,
        input.parser_version ?? null,
        input.summary ?? null,
        stringifyJson(input.tags ?? []),
        stringifyJson(input.metadata ?? null),
        timestamp,
        existing.id,
      );
      return this.getSource(existing.id)!;
    }

    const id = nanoid(16);
    db.prepare(
      `INSERT INTO knowledge_sources (
        id, project_id, room_id, source_type, source_id, title, mime_type, size, status,
        content_hash, parser, parser_version, summary, tags_json, metadata_json,
        last_processed_at, created_at, updated_at, deleted_at
      ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, NULL, ?, ?, NULL)`,
    ).run(
      id,
      input.project_id,
      input.room_id ?? null,
      input.source_type,
      input.source_id,
      input.title,
      input.mime_type ?? null,
      input.size ?? null,
      input.status,
      input.content_hash ?? null,
      input.parser ?? null,
      input.parser_version ?? null,
      input.summary ?? null,
      stringifyJson(input.tags ?? []),
      stringifyJson(input.metadata ?? null),
      timestamp,
      timestamp,
    );
    return this.getSource(id)!;
  },

  getSource(id: string): KnowledgeSourceListItem | undefined {
    const source = db.prepare(
      `SELECT
         knowledge_sources.*,
         (SELECT COUNT(1) FROM knowledge_chunks WHERE knowledge_chunks.source_id = knowledge_sources.id AND enabled = 1) AS chunk_count,
         (SELECT error FROM knowledge_extractions WHERE knowledge_extractions.source_id = knowledge_sources.id ORDER BY created_at DESC LIMIT 1) AS extraction_error
       FROM knowledge_sources
       WHERE id = ? AND deleted_at IS NULL`,
    ).get(id) as (KnowledgeSource & { chunk_count: number; extraction_error: string | null }) | undefined;
    return source ? hydrateSource(source) : undefined;
  },

  listSources(filters: SourceFilters): KnowledgeSourceListItem[] {
    const where = ['knowledge_sources.project_id = @projectId', 'knowledge_sources.deleted_at IS NULL'];
    const params: Record<string, unknown> = { projectId: filters.projectId };
    if (filters.roomId) {
      where.push('knowledge_sources.room_id = @roomId');
      params.roomId = filters.roomId;
    }
    if (filters.statuses?.length) {
      where.push(`knowledge_sources.status IN (${filters.statuses.map((_, index) => `@status${index}`).join(', ')})`);
      filters.statuses.forEach((status, index) => {
        params[`status${index}`] = status;
      });
    }
    if (filters.sourceTypes?.length) {
      where.push(`knowledge_sources.source_type IN (${filters.sourceTypes.map((_, index) => `@sourceType${index}`).join(', ')})`);
      filters.sourceTypes.forEach((type, index) => {
        params[`sourceType${index}`] = type;
      });
    }
    const query = normalizeLikeQuery(filters.query);
    if (query) {
      where.push(`(
        knowledge_sources.title LIKE @query ESCAPE '\\'
        OR COALESCE(knowledge_sources.summary, '') LIKE @query ESCAPE '\\'
        OR COALESCE(knowledge_sources.tags_json, '') LIKE @query ESCAPE '\\'
      )`);
      params.query = query;
    }
    params.limit = Math.max(1, Math.min(filters.limit ?? 100, 200));
    return db.prepare(
      `SELECT
         knowledge_sources.*,
         (SELECT COUNT(1) FROM knowledge_chunks WHERE knowledge_chunks.source_id = knowledge_sources.id AND enabled = 1) AS chunk_count,
         (SELECT error FROM knowledge_extractions WHERE knowledge_extractions.source_id = knowledge_sources.id ORDER BY created_at DESC LIMIT 1) AS extraction_error
       FROM knowledge_sources
       WHERE ${where.join(' AND ')}
       ORDER BY knowledge_sources.updated_at DESC
       LIMIT @limit`,
    ).all(params).map((row) => hydrateSource(row as KnowledgeSource & { chunk_count: number; extraction_error: string | null }));
  },

  saveExtraction(input: SaveExtractionInput): KnowledgeExtraction {
    const id = nanoid(16);
    db.prepare(
      `INSERT INTO knowledge_extractions (
        id, source_id, plain_text, markdown, layout_json, table_json, image_json, error, created_at
      ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)`,
    ).run(
      id,
      input.source_id,
      input.plain_text ?? null,
      input.markdown ?? null,
      stringifyJson(input.layout ?? null),
      stringifyJson(input.table ?? null),
      stringifyJson(input.image ?? null),
      input.error ?? null,
      now(),
    );
    return db.prepare('SELECT * FROM knowledge_extractions WHERE id = ?').get(id) as KnowledgeExtraction;
  },

  getLatestExtraction(sourceId: string): KnowledgeExtraction | undefined {
    return db.prepare(
      `SELECT * FROM knowledge_extractions WHERE source_id = ? ORDER BY created_at DESC LIMIT 1`,
    ).get(sourceId) as KnowledgeExtraction | undefined;
  },

  replaceChunks(sourceId: string, chunks: ReplaceChunkInput[]): KnowledgeChunk[] {
    const timestamp = now();
    const transaction = db.transaction(() => {
      const existingChunkIds = db.prepare('SELECT id FROM knowledge_chunks WHERE source_id = ?')
        .all(sourceId)
        .map((row) => (row as { id: string }).id);

        db.prepare(`DELETE FROM knowledge_chunk_fts WHERE chunk_id IN (${placeholders})`).run(...existingChunkIds);
      }
      db.prepare('DELETE FROM knowledge_chunks WHERE source_id = ?').run(sourceId);
      const inserted: KnowledgeChunk[] = [];
      const chunkStmt = db.prepare(
        `INSERT INTO knowledge_chunks (
          id, source_id, project_id, room_id, chunk_index, chunk_type, title, content, summary,
          metadata_json, page_start, page_end, token_estimate, content_hash, enabled, created_at, updated_at
        ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)`,
      );
      const ftsStmt = db.prepare(
        `INSERT INTO knowledge_chunk_fts (chunk_id, source_id, title, content, summary, tags)
         VALUES (?, ?, ?, ?, ?, ?)`,
      );
      for (const chunk of chunks) {

        inserted.push(db.prepare('SELECT * FROM knowledge_chunks WHERE id = ?').get(id) as KnowledgeChunk);
      }

    });

  },

    status: KnowledgeStatus;
    summary?: string | null;
    tags?: string[];
    metadata?: JsonRecord | null;
    last_processed_at?: number | null;
  }): void {
    db.prepare(
      `UPDATE knowledge_sources
       SET status = ?, summary = COALESCE(?, summary), tags_json = COALESCE(?, tags_json),
           metadata_json = COALESCE(?, metadata_json), last_processed_at = COALESCE(?, last_processed_at),
           updated_at = ?
       WHERE id = ?`,
    ).run(
      input.status,
      input.summary ?? null,
      input.tags ? stringifyJson(input.tags) : null,
      input.metadata !== undefined ? stringifyJson(input.metadata) : null,
      input.last_processed_at ?? null,

  search(input: SearchInput): KnowledgeSearchResult[] {
    const where = [
      'knowledge_sources.project_id = @projectId',
      'knowledge_sources.deleted_at IS NULL',
      'knowledge_chunks.enabled = 1',
    ];
    const params: Record<string, unknown> = {
      projectId: input.projectId,
      query: input.query,
      limit: Math.max(1, Math.min(input.limit ?? 10, 50)),
    };
    if (input.roomId) {
      where.push('(knowledge_sources.room_id = @roomId OR knowledge_sources.room_id IS NULL)');
      params.roomId = input.roomId;
    }
    if (input.statuses?.length) {
      where.push(`knowledge_sources.status IN (${input.statuses.map((_, index) => `@status${index}`).join(', ')})`);
      input.statuses.forEach((status, index) => {
        params[`status${index}`] = status;
      });
    } else {
      where.push("knowledge_sources.status = 'ready'");
    }
    if (input.sourceTypes?.length) {
      where.push(`knowledge_sources.source_type IN (${input.sourceTypes.map((_, index) => `@sourceType${index}`).join(', ')})`);
      input.sourceTypes.forEach((type, index) => {
        params[`sourceType${index}`] = type;
      });
    }
    const rows = db.prepare(
      `SELECT
         knowledge_sources.id AS source_id,
         knowledge_chunks.id AS chunk_id,
         knowledge_chunks.title,
         snippet(knowledge_chunk_fts, 3, '[', ']', '…', 12) AS snippet,
         bm25(knowledge_chunk_fts) AS score,
         knowledge_sources.source_type,
         knowledge_sources.mime_type,
         knowledge_sources.status,
         knowledge_chunks.page_start,
         knowledge_chunks.page_end,
         knowledge_sources.tags_json
       FROM knowledge_chunk_fts
       JOIN knowledge_chunks ON knowledge_chunks.id = knowledge_chunk_fts.chunk_id
       JOIN knowledge_sources ON knowledge_sources.id = knowledge_chunks.source_id
       WHERE knowledge_chunk_fts MATCH @query
         AND ${where.join(' AND ')}
       ORDER BY score ASC
       LIMIT @limit`,
    ).all(params) as Array<{
      source_id: string;
      chunk_id: string;
      title: string;
      snippet: string;
      score: number;
      source_type: KnowledgeSourceType;
      mime_type: string | null;
      status: KnowledgeStatus;
      page_start: number | null;
      page_end: number | null;
      tags_json: string | null;
    }>;
    return rows.map((row) => ({
      source_id: row.source_id,
      chunk_id: row.chunk_id,
      title: row.title,
      snippet: row.snippet,
      score: row.score,
      source_type: row.source_type,
      mime_type: row.mime_type,
      status: row.status,
      page_start: row.page_start,
      page_end: row.page_end,
      tags: parseJsonArray(row.tags_json),
      citation: `${row.title}#${row.chunk_id}`,
    }));
  },
};

function hydrateSource(row: KnowledgeSource & { chunk_count: number; extraction_error: string | null }): KnowledgeSourceListItem {
  return {
    ...row,
    tags: parseJsonArray(row.tags_json),
    metadata: parseJsonRecord(row.metadata_json),
    chunk_count: row.chunk_count,
    extraction_error: row.extraction_error,
  };
}

function stringifyJson(value: unknown): string | null {
  if (value === null || value === undefined) return null;
  return JSON.stringify(value);
}

function parseJsonArray(value: string | null): string[] {
  if (!value) return [];
  try {
    const parsed = JSON.parse(value) as unknown;
    return Array.isArray(parsed) ? parsed.filter((item): item is string => typeof item === 'string') : [];
  } catch {
    return [];
  }
}

function parseJsonRecord(value: string | null): JsonRecord | null {
  if (!value) return null;
  try {
    const parsed = JSON.parse(value) as unknown;
    return parsed && typeof parsed === 'object' && !Array.isArray(parsed) ? parsed as JsonRecord : null;
  } catch {
    return null;
  }
}

function metadataTags(value: JsonRecord | null | undefined): string {
  const tags = value?.tags;
  return Array.isArray(tags) ? tags.filter((item): item is string => typeof item === 'string').join(' ') : '';
}

function normalizeLikeQuery(value: string | undefined): string | null {
  const trimmed = value?.trim();
  if (!trimmed) return null;
  return `%${trimmed.replace(/[\\%_]/g, (char) => `\\${char}`)}%`;
}
```

```ts
import { createHash } from 'node:crypto';
import type { KnowledgeChunkType } from './knowledge-types.js';

const TEXT_MIME_MARKERS = ['text/', 'json', 'xml', 'yaml', 'csv', 'markdown'];
const TEXT_EXTENSIONS = ['.txt', '.md', '.markdown', '.json', '.csv', '.yaml', '.yml', '.xml'];

export interface KnowledgeExtractInput {
  title: string;
  mimeType: string | null;
  content: string | null;
}

export interface KnowledgeExtractResult {
  parser: string;
  parserVersion: string;
  plainText: string;
  markdown: string | null;
  layout: Record<string, unknown> | null;
  table: Record<string, unknown> | null;
  image: Record<string, unknown> | null;
  contentHash: string;
}

export interface KnowledgeSummaryResult {
  summary: string;
  tags: string[];
  keyPoints: string[];
  contentKind: string;
}

export interface KnowledgeChunkDraft {
  chunk_index: number;
  chunk_type: KnowledgeChunkType;
  title: string;
  content: string;
  summary: string | null;
  metadata: Record<string, unknown>;
  page_start: number | null;
  page_end: number | null;
  token_estimate: number;
  content_hash: string;
  enabled: 0 | 1;
}

export async function extractKnowledgeText(input: KnowledgeExtractInput): Promise<KnowledgeExtractResult> {
  const mimeType = (input.mimeType ?? '').toLowerCase();
  if (mimeType.startsWith('image/')) {
    return {
      parser: 'image-metadata',
      parserVersion: '1',
      plainText: '',
      markdown: null,
      layout: null,
      table: null,
      image: { kind: 'image', title: input.title, mimeType: input.mimeType },
      contentHash: hashText(`${input.title}:${input.mimeType ?? ''}`),
    };
  }
  if (!isTextLike(input.title, mimeType) || input.content === null) {
    return {
      parser: 'metadata-only',
      parserVersion: '1',
      plainText: '',
      markdown: null,
      layout: null,
      table: null,
      image: null,
      contentHash: hashText(`${input.title}:${input.mimeType ?? ''}`),
    };
  }
  const normalized = normalizeText(input.content);
  return {
    parser: 'builtin-text',
    parserVersion: '1',
    plainText: normalized,
    markdown: mimeType.includes('markdown') || input.title.toLowerCase().endsWith('.md') ? normalized : null,
    layout: null,
    table: mimeType.includes('csv') || input.title.toLowerCase().endsWith('.csv') ? { kind: 'csv' } : null,
    image: null,
    contentHash: hashText(normalized),
  };
}

export function summarizeKnowledgeText(text: string, title: string): KnowledgeSummaryResult {
  const lines = text.split(/\n+/).map((line) => line.trim()).filter(Boolean);
  const keyPoints = lines.slice(0, 5).map((line) => line.replace(/^#+\s*/, '').slice(0, 160));
  const summaryBase = keyPoints.length > 0 ? keyPoints.join('；') : `${title} 暂无可提取文本内容`;
  return {
    summary: summaryBase.slice(0, 300),
    tags: deriveTags(`${title}\n${text}`).slice(0, 8),
    keyPoints,
    contentKind: inferContentKind(title, text),
  };
}

export function splitKnowledgeChunks(input: {
  title: string;
  text: string;
  maxChars?: number;
}): KnowledgeChunkDraft[] {
  const maxChars = Math.max(12, input.maxChars ?? 1800);
  const text = normalizeText(input.text);
  if (!text) return [];
  const chunks: KnowledgeChunkDraft[] = [];
  let offset = 0;
  while (offset < text.length) {
    const slice = text.slice(offset, offset + maxChars);
    chunks.push({
      chunk_index: chunks.length,
      chunk_type: 'body',
      title: input.title,
      content: slice,
      summary: slice.slice(0, 160),
      metadata: { tags: deriveTags(slice).slice(0, 5) },
      page_start: null,
      page_end: null,
      token_estimate: estimateTokens(slice),
      content_hash: hashText(slice),
      enabled: 1,
    });
    offset += maxChars;
  }
  return chunks;
}

export function hashText(value: string): string {
  return createHash('sha256').update(value).digest('hex');
}

function isTextLike(title: string, mimeType: string): boolean {
  return TEXT_MIME_MARKERS.some((marker) => mimeType.includes(marker)) ||
    TEXT_EXTENSIONS.some((extension) => title.toLowerCase().endsWith(extension));
}

function normalizeText(value: string): string {
  return value.replace(/\r\n/g, '\n').replace(/\u0000/g, '').trim();
}

function estimateTokens(value: string): number {
  return Math.ceil(value.length / 4);
}

function deriveTags(value: string): string[] {
  const candidates = [
    ['Markdown', /markdown|\.md/i],
    ['图片', /image|png|jpg|截图|图片/i],
    ['表格', /csv|xlsx|表格|清单/i],
    ['设计', /设计|布局|UI|视觉/i],
    ['验收', /验收|测试|验证/i],
    ['计划', /计划|任务|路线/i],
    ['风险', /风险|失败|错误/i],
    ['知识库', /知识库|资源库|索引/i],
  ] as const;
  const tags = candidates.filter(([, pattern]) => pattern.test(value)).map(([tag]) => tag);
  return tags.length > 0 ? [...new Set(tags)] : ['资料'];
}

function inferContentKind(title: string, text: string): string {
  const corpus = `${title}\n${text}`;
  if (/验收|测试|验证/.test(corpus)) return '验收';
  if (/设计|布局|UI|视觉/.test(corpus)) return '设计';
  if (/风险|约束|决定|结论/.test(corpus)) return '决策';
  if (/清单|csv|xlsx|表格/i.test(corpus)) return '表格';
  return '资料';
}
```

```ts
import { readFile } from 'node:fs/promises';
import type { ProjectFile } from './types.js';
import { knowledgeRepo } from './repos/knowledge.js';
import {
  extractKnowledgeText,
  splitKnowledgeChunks,
  summarizeKnowledgeText,
} from './knowledge-extraction.js';

export async function ingestProjectFileIntoKnowledge(file: ProjectFile): Promise<ReturnType<typeof knowledgeRepo.getSource> extends infer T ? NonNullable<T> : never> {
  const sourceType = file.source_type === 'agent_document' ? 'agent_document' : 'uploaded_file';
  const rawContent = await readProjectFileContent(file);
  const extracted = await extractKnowledgeText({
    title: file.original_name,
    mimeType: file.mime_type,
    content: rawContent,
  });
  const source = knowledgeRepo.ensureSource({
    project_id: file.project_id,
    room_id: file.source_room_id,
    source_type: sourceType,
    source_id: normalizeSourceId(file.id),
    title: file.original_name,
    mime_type: file.mime_type,
    size: file.size,
    status: 'processing',
    content_hash: extracted.contentHash,
    parser: extracted.parser,
    parser_version: extracted.parserVersion,
    summary: null,
    tags: [],
    metadata: {
      file_id: file.id,
      source_label: file.source_label,
      source_agent_id: file.source_agent_id,
      source_task_id: file.source_task_id,
    },
  });

  try {
    knowledgeRepo.saveExtraction({
      source_id: source.id,
      plain_text: extracted.plainText,
      markdown: extracted.markdown,
      layout: extracted.layout,
      table: extracted.table,
      image: extracted.image,
      error: null,
    });
    const summary = summarizeKnowledgeText(extracted.plainText, file.original_name);
    const chunks = splitKnowledgeChunks({ title: file.original_name, text: extracted.plainText });
    knowledgeRepo.replaceChunks(source.id, chunks.map((chunk) => ({
      ...chunk,
      project_id: file.project_id,
      room_id: file.source_room_id,
    })));
    knowledgeRepo.updateSourceStatus(source.id, {
      status: 'ready',
      summary: summary.summary,
      tags: summary.tags,
      metadata: {
        key_points: summary.keyPoints,
        content_kind: summary.contentKind,
        file_id: file.id,
      },
      last_processed_at: Date.now(),
    });
  } catch (error) {
    knowledgeRepo.saveExtraction({
      source_id: source.id,
      error: error instanceof Error ? error.message : 'knowledge ingestion failed',
    });
    knowledgeRepo.updateSourceStatus(source.id, {
      status: 'failed',
      metadata: { error: error instanceof Error ? error.message : 'knowledge ingestion failed' },
      last_processed_at: Date.now(),
    });
  }

  return knowledgeRepo.getSource(source.id)!;
}

async function readProjectFileContent(file: ProjectFile): Promise<string | null> {
  if (file.source_type === 'agent_document') return file.content ?? '';
  if (!isTextLikeFile(file) || !file.storage_path) return null;
  return readFile(file.storage_path, 'utf8');
}

function isTextLikeFile(file: ProjectFile): boolean {
  const mimeType = file.mime_type.toLowerCase();
  return mimeType.startsWith('text/') ||
    mimeType.includes('json') ||
    mimeType.includes('xml') ||
    mimeType.includes('yaml') ||
    mimeType.includes('csv') ||
    /\.(txt|md|markdown|json|csv|yaml|yml|xml)$/i.test(file.original_name);
}

function normalizeSourceId(id: string): string {
  return id.startsWith('asset:') ? id.slice('asset:'.length) : id;
}
```

```ts
function toResponseHeaders(headers: OutgoingHttpHeaders): Headers {
  const responseHeaders = new Headers();
  for (const [name, value] of Object.entries(headers)) {
    if (value === undefined) continue;
    if (Array.isArray(value)) {
      for (const item of value) responseHeaders.append(name, item);
    } else {
      responseHeaders.set(name, String(value));
    }
  }

async function request(path: string, init: RequestInit = {}): Promise<Response> {
  const serializedRequest = new Request(`http://127.0.0.1${path}`, init);
  const body = init.body === undefined || init.body === null
    ? null
    : Buffer.from(await serializedRequest.arrayBuffer());
  const socket = new InMemorySocket();
  const req = new IncomingMessage(socket as unknown as import('node:net').Socket);
  req.method = init.method ?? 'GET';
  req.url = path;
  req.headers = Object.fromEntries(serializedRequest.headers);
  req.httpVersion = '1.1';
  req.httpVersionMajor = 1;
  req.httpVersionMinor = 1;
  if (body) req.headers['content-length'] = String(body.byteLength);
  const res = new ServerResponse(req);
  res.assignSocket(socket as unknown as import('node:net').Socket);
  const chunks: Buffer[] = [];
  res.write = ((chunk: unknown, encoding?: BufferEncoding | ((error?: Error | null) => void), callback?: (error?: Error | null) => void) => {
    if (chunk) chunks.push(Buffer.isBuffer(chunk) ? chunk : Buffer.from(String(chunk), typeof encoding === 'string' ? encoding : undefined));
    if (typeof encoding === 'function') encoding();
    if (callback) callback();
    return true;
  }) as typeof res.write;
  res.end = ((chunk?: unknown, encoding?: BufferEncoding | (() => void), callback?: () => void) => {
    if (chunk) chunks.push(Buffer.isBuffer(chunk) ? chunk : Buffer.from(String(chunk), typeof encoding === 'string' ? encoding : undefined));
    if (typeof encoding === 'function') encoding();
    if (callback) callback();
    res.emit('finish');
    res.emit('close');
    return res;
  }) as typeof res.end;
  const responsePromise = new Promise<Response>((resolve, reject) => {
    res.once('finish', () => {
      const responseBody = res.statusCode === 204 || res.statusCode === 304 ? null : Buffer.concat(chunks);
      resolve(new Response(responseBody, {
        status: res.statusCode,
        headers: toResponseHeaders(res.getHeaders()),
      }));
    });
    (app as unknown as { handle: (...args: unknown[]) => void }).handle(req, res, (error: unknown) => {
      if (error) reject(error);
    });
  });
  if (body) req.push(body);
  req.push(null);
  req.complete = true;
  return responsePromise;
}
```

```ts
function parseCsvEnum<T extends string>(value: string | undefined, schema: z.ZodEnum<[T, ...T[]]>): T[] | undefined {
  if (!value) return undefined;
  const parsed: T[] = [];
  for (const item of value.split(',').map((part) => part.trim()).filter(Boolean)) {
    const result = schema.safeParse(item);
    if (result.success) parsed.push(result.data);
  }
```

```ts
function renderKnowledgeContextBlock(
  projectId: string,
  file: ProjectFile,
  budget: number,
): { block: string; usedChars: number } | null {
  if (budget <= 0) return null;
  const sourceType = file.source_type === 'agent_document' ? 'agent_document' : 'uploaded_file';
  const sourceId = file.source_type === 'agent_document' && file.id.startsWith('asset:')
    ? file.id.slice('asset:'.length)
    : file.id;
  const source = knowledgeRepo.listSources({
    projectId,
    sourceTypes: [sourceType],
    statuses: ['ready'],
    query: file.original_name,
    limit: 20,
  }).find((item) => item.source_id === sourceId);
  if (!source) return null;
  const chunks = knowledgeRepo.listChunks(source.id).filter((chunk) => chunk.enabled === 1);
  if (chunks.length === 0) return null;
  const parts: string[] = [];
  let usedChars = 0;
  for (const chunk of chunks) {
    if (usedChars >= budget) break;
    const limit = Math.min(MAX_KNOWLEDGE_CHUNK_CHARS, budget - usedChars);
    const content = chunk.content.slice(0, limit);
    usedChars += content.length;
    parts.push([
      `#### Chunk ${chunk.chunk_index + 1}`,
      content,
      `Citation: ${source.title}#${chunk.id}`,
    ].join('\n'));
  }
  if (parts.length === 0) return null;
  return {
    usedChars,
    block: [
      `### Knowledge: ${source.title}`,
      `- Source type: ${source.source_type}`,
      `- Status: ${source.status}`,
      `- Summary: ${source.summary ?? '(none)'}`,
      '',
      ...parts,
    ].join('\n'),
  };
}
```

```ts
export function getKnowledgeStatusLabel(status: KnowledgeStatus, locale: KnowledgeLocale): string {
  const labels: Record<KnowledgeStatus, { zh: string; en: string }> = {
    pending: { zh: '待索引', en: 'Pending' },
    processing: { zh: 'AI 分析中', en: 'Processing' },
    ready: { zh: '已提取', en: 'Extracted' },
    failed: { zh: '失败', en: 'Failed' },
    disabled: { zh: '已禁用', en: 'Disabled' },
    stale: { zh: '已过期', en: 'Stale' },
  };
  return labels[status][locale];
}

export function getKnowledgeStatusTone(status: KnowledgeStatus): 'info' | 'success' | 'warning' | 'danger' | 'muted' {
  if (status === 'ready') return 'success';
  if (status === 'processing' || status === 'pending') return 'info';
  if (status === 'stale') return 'warning';
  if (status === 'failed') return 'danger';
  return 'muted';
}

export function knowledgeSourceMatchesQuery(source: KnowledgeSourceListItem, query: string): boolean {
  const normalized = query.trim().toLowerCase();
  if (!normalized) return true;
  return [
    source.title,
    source.summary,
    source.mime_type,
    source.source_type,
    source.status,
    ...source.tags,
  ].some((value) => value?.toLowerCase().includes(normalized));
}
```
