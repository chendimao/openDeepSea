# 会话中间区文件浏览器标签页 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 将会话页中间区域升级为外层固定“对话记录 / 文件浏览器”tabs，并在文件浏览器内实现目录树、可关闭文件 tabs、Monaco 只读文本预览、图片预览与文档扩展点。

**Architecture:** 保留 `SessionShellView` 的三栏工作台，只用 `SessionCenterWorkspace` 替换中间 `TranscriptCanvas` 位置。外层 `flexlayout-react` 管两个固定 tab；`FileBrowserWorkspace` 内左侧使用 `react-complex-tree`，右侧使用内层 `flexlayout-react` 管文件 tabs；viewer registry 按文件类型分发到 Monaco、图片或 unsupported viewer。后端新增只读图片 blob 接口，复用 workspace path 安全边界。

**Tech Stack:** React 18、TypeScript、Vite、TanStack Query、Express、Node test、`flexlayout-react`、`react-complex-tree`、`monaco-editor`、`@monaco-editor/react`。

```ts
async function captureApiRequest<T>(call: () => Promise<T>): Promise<string> {
  let requestedUrl = '';
  const originalFetch = globalThis.fetch;
  globalThis.fetch = (async (input: RequestInfo | URL) => {
    requestedUrl = String(input);
    return new Response(JSON.stringify({ path: '', entries: [] }), {
      status: 200,
      headers: { 'Content-Type': 'application/json' },
    });
  }) as typeof fetch;
  try {
    await call();
    return requestedUrl;
  } finally {
    globalThis.fetch = originalFetch;
  }
}
```

```ts
export async function readWorkspaceImageBlob(
  projectPath: string,
  inputPath: string,
): Promise<{ path: string; mimeType: string; bytes: Buffer }> {
  const resolved = await resolveWorkspacePath(projectPath, inputPath);
  ensureWorkspacePathAllowed(resolved.relativePath, resolved.symlinkTargetRelativePath);

  const fileHandle = await openWorkspaceFileForRead(resolved.absolutePath);
  try {
    const fileStats = await fileHandle.stat();
    if (!fileStats.isFile()) {
      throw workspaceFileError('WORKSPACE_PATH_NOT_FILE');
    }
    if (fileStats.size > WORKSPACE_IMAGE_BLOB_SIZE_LIMIT) {
      throw workspaceFileError('WORKSPACE_FILE_TOO_LARGE');
    }

    const mimeType = detectMimeType(resolved.relativePath);
    if (!mimeType.startsWith('image/')) {
      throw workspaceFileError('WORKSPACE_FILE_BINARY');
    }

    const bytes = await fileHandle.readFile();
    return {
      path: resolved.relativePath,
      mimeType,
      bytes,
    };
  } finally {
    await fileHandle.close();
  }
}
```

```ts
import type {
  WorkspaceDirectoryEntry,
  WorkspaceFileViewerKind,
} from '../lib/types';

export interface WorkspaceFileTab {
  id: string;
  path: string;
  name: string;
  mimeType: string | null;
  language: string | null;
  viewer: WorkspaceFileViewerKind;
}

export function workspaceFileTabId(path: string): string {
  return `file:${path}`;
}

export function resolveWorkspaceFileViewer(file: Pick<WorkspaceDirectoryEntry, 'name' | 'mimeType' | 'language'>): WorkspaceFileViewerKind {
  const mimeType = file.mimeType?.toLowerCase() ?? '';
  if (mimeType.startsWith('image/')) return 'image';
  if (file.language) return 'text';
  if (mimeType.startsWith('text/') || mimeType.includes('json') || mimeType.includes('xml')) return 'text';
  const extension = file.name.split('.').pop()?.toLowerCase() ?? '';
  if (['ts', 'tsx', 'js', 'jsx', 'json', 'md', 'css', 'html', 'txt', 'yml', 'yaml', 'sh', 'py', 'go', 'rs'].includes(extension)) return 'text';
  return 'unsupported';
}

export function createWorkspaceFileTab(file: WorkspaceDirectoryEntry): WorkspaceFileTab {
  return {
    id: workspaceFileTabId(file.path),
    path: file.path,
    name: file.name,
    mimeType: file.mimeType,
    language: file.language,
    viewer: resolveWorkspaceFileViewer(file),
  };
}

export function openWorkspaceFileTab(
  tabs: WorkspaceFileTab[],
  file: WorkspaceDirectoryEntry,
): { tabs: WorkspaceFileTab[]; activePath: string } {
  const existing = tabs.find((tab) => tab.path === file.path);
  if (existing) return { tabs, activePath: existing.path };
  const next = createWorkspaceFileTab(file);
  return { tabs: [...tabs, next], activePath: next.path };
}

export function closeWorkspaceFileTab(
  tabs: WorkspaceFileTab[],
  path: string,
): { tabs: WorkspaceFileTab[]; activePath: string | null } {
  const index = tabs.findIndex((tab) => tab.path === path);
  if (index < 0) return { tabs, activePath: tabs[0]?.path ?? null };
  const nextTabs = tabs.filter((tab) => tab.path !== path);
  const fallback = nextTabs[Math.max(0, index - 1)] ?? nextTabs[0] ?? null;
  return { tabs: nextTabs, activePath: fallback?.path ?? null };
}

export function reorderWorkspaceFileTabs(tabs: WorkspaceFileTab[], ids: string[]): WorkspaceFileTab[] {
  const byId = new Map(tabs.map((tab) => [tab.id, tab]));
  const ordered = ids.flatMap((id) => {
    const tab = byId.get(id);
    return tab ? [tab] : [];
  });
  const included = new Set(ordered.map((tab) => tab.id));
  return [...ordered, ...tabs.filter((tab) => !included.has(tab.id))];
}
```

```tsx
import { useQuery } from '@tanstack/react-query';
import React, { useMemo, useState } from 'react';
import { StaticTreeDataProvider, Tree, UncontrolledTreeEnvironment } from 'react-complex-tree';
import 'react-complex-tree/lib/style-modern.css';
import { api } from '../lib/api';
import type { WorkspaceDirectoryEntry } from '../lib/types';

type TreeItemData = WorkspaceDirectoryEntry & { title: string };

export function WorkspaceFileTree({
  projectId,
  onOpenFile,
}: {
  projectId: string;
  onOpenFile: (file: WorkspaceDirectoryEntry) => void;
}): JSX.Element {
  const [loadedDirs, setLoadedDirs] = useState<Record<string, WorkspaceDirectoryEntry[]>>({});
  const rootQuery = useQuery({
    queryKey: ['workspace-tree', projectId, ''],
    queryFn: () => api.listWorkspaceDirectory(projectId, ''),
  });

  const rootEntries = rootQuery.data?.entries ?? [];
  const treeItems = useMemo(() => buildTreeItems(rootEntries, loadedDirs), [rootEntries, loadedDirs]);

  const dataProvider = useMemo(
    () => new StaticTreeDataProvider(treeItems, (item, data) => ({ ...item, data })),
    [treeItems],
  );

  if (rootQuery.isLoading) return <div className="deepsea-file-tree-state">正在加载目录...</div>;
  if (rootQuery.isError) {
    return (
      <div className="deepsea-file-tree-state">
        <strong>目录加载失败</strong>
        <button type="button" onClick={() => void rootQuery.refetch()}>重试</button>
      </div>
    );
  }

  return (
    <UncontrolledTreeEnvironment<TreeItemData>
      dataProvider={dataProvider}
      getItemTitle={(item) => item.data.title}
      viewState={{}}
      canDragAndDrop={false}
      canDropOnFolder={false}
      canReorderItems={false}
      onSelectItems={(items) => {
        const item = items[0] ? treeItems[String(items[0])] : null;
        if (item?.data.type === 'file') onOpenFile(item.data);
      }}
      onExpandItem={(item) => {
        if (item.data.type !== 'directory') return;
        void api.listWorkspaceDirectory(projectId, item.data.path).then((result) => {
          setLoadedDirs((current) => ({ ...current, [item.data.path]: result.entries }));
        });
      }}
    >
      <Tree treeId="workspace-tree" rootItem="root" treeLabel="当前项目目录" />
    </UncontrolledTreeEnvironment>
  );
}

function buildTreeItems(
  rootEntries: WorkspaceDirectoryEntry[],
  loadedDirs: Record<string, WorkspaceDirectoryEntry[]>,
) {
  const items: Record<string, { index: string; canMove: boolean; isFolder: boolean; children: string[]; data: TreeItemData }> = {
    root: {
      index: 'root',
      canMove: false,
      isFolder: true,
      children: rootEntries.map((entry) => entry.path),
      data: {
        name: 'root',
        title: 'root',
        path: '',
        type: 'directory',
        size: null,
        mimeType: null,
        language: null,
      },
    },
  };

  const appendEntry = (entry: WorkspaceDirectoryEntry) => {
    const children = loadedDirs[entry.path] ?? [];
    items[entry.path] = {
      index: entry.path,
      canMove: false,
      isFolder: entry.type === 'directory',
      children: children.map((child) => child.path),
      data: { ...entry, title: entry.name },
    };
    children.forEach(appendEntry);
  };
  rootEntries.forEach(appendEntry);
  return items;
}
```

```tsx
import { Actions, DockLocation, Layout, Model, type TabNode } from 'flexlayout-react';
import React, { useEffect, useMemo, useRef } from 'react';
import type { WorkspaceFileTab } from './workspace-file-model';
import { FileViewer } from './FileViewers';

const FILE_TABSET_ID = 'workspace-file-tabs';

export function WorkspaceFileTabs({
  projectId,
  tabs,
  activePath,
  onClose,
  onReorder,
  onFocus,
}: {
  projectId: string;
  tabs: WorkspaceFileTab[];
  activePath: string | null;
  onClose: (path: string) => void;
  onReorder: (ids: string[]) => void;
  onFocus: (path: string) => void;
}): JSX.Element {
  const modelRef = useRef<Model | null>(null);
  const model = useMemo(() => {
    const next = createFileTabsModel(tabs, activePath);
    modelRef.current = next;
    return next;
  }, [tabs, activePath]);

  useEffect(() => {
    if (!activePath || !modelRef.current) return;
    modelRef.current.doAction(Actions.selectTab(`file:${activePath}`));
  }, [activePath]);

  if (tabs.length === 0) {
    return <div className="deepsea-file-tabs-empty">从左侧目录选择文件</div>;
  }

  return (
    <div className="deepsea-file-tabs">
      <Layout
        model={model}
        factory={(node) => {
          const path = node.getConfig()?.path as string | undefined;
          const tab = tabs.find((item) => item.path === path);
          return tab ? <FileViewer projectId={projectId} tab={tab} /> : <div className="deepsea-file-viewer-state">文件已关闭</div>;
        }}
        onAction={(action) => {
          if (action.type === Actions.DELETE_TAB) {
            const node = model.getNodeById(String(action.data.node));
            const path = node?.getConfig()?.path as string | undefined;
            if (path) onClose(path);
            return undefined;
          }
          if (action.type === Actions.SELECT_TAB) {
            const node = model.getNodeById(String(action.data.tabNode));
            const path = node?.getConfig()?.path as string | undefined;
            if (path) onFocus(path);
          }
          return action;
        }}
        onModelChange={(nextModel) => {
          const tabset = nextModel.getNodeById(FILE_TABSET_ID);
          const ids = tabset && 'getChildren' in tabset
            ? tabset.getChildren().map((node) => node.getId())
            : tabs.map((tab) => tab.id);
          onReorder(ids);
        }}
        onRenderTab={(node: TabNode, renderValues) => {
          renderValues.content = <span title={String(node.getConfig()?.path ?? node.getName())}>{node.getName()}</span>;
        }}
      />
    </div>
  );
}

function createFileTabsModel(tabs: WorkspaceFileTab[], activePath: string | null): Model {
  const selected = Math.max(0, tabs.findIndex((tab) => tab.path === activePath));
  return Model.fromJson({
    global: {
      tabSetEnableClose: false,
      tabEnableRename: false,
      tabSetEnableMaximize: false,
      enableEdgeDock: false,
    },
    borders: [],
    layout: {
      type: 'row',
      children: [{
        type: 'tabset',
        id: FILE_TABSET_ID,
        selected,
        enableDeleteWhenEmpty: false,
        children: tabs.map((tab) => ({
          type: 'tab',
          id: tab.id,
          name: tab.name,
          component: 'workspace-file',
          enableClose: true,
          config: { path: tab.path },
        })),
      }],
    },
  });
}
```

```tsx
import React, { useState } from 'react';
import type { WorkspaceDirectoryEntry } from '../lib/types';
import { WorkspaceFileTree } from './WorkspaceFileTree';
import { WorkspaceFileTabs } from './WorkspaceFileTabs';
import {
  closeWorkspaceFileTab,
  openWorkspaceFileTab,
  reorderWorkspaceFileTabs,
  type WorkspaceFileTab,
} from './workspace-file-model';

export function FileBrowserWorkspace({ projectId }: { projectId: string }): JSX.Element {
  const [tabs, setTabs] = useState<WorkspaceFileTab[]>([]);
  const [activePath, setActivePath] = useState<string | null>(null);

  const openFile = (file: WorkspaceDirectoryEntry) => {
    if (file.type !== 'file') return;
    const result = openWorkspaceFileTab(tabs, file);
    setTabs(result.tabs);
    setActivePath(result.activePath);
  };

  const closeFile = (path: string) => {
    const result = closeWorkspaceFileTab(tabs, path);
    setTabs(result.tabs);
    setActivePath(result.activePath);
  };

  return (
    <div className="deepsea-file-browser">
      <aside className="deepsea-file-browser__tree">
        <WorkspaceFileTree projectId={projectId} onOpenFile={openFile} />
      </aside>
      <section className="deepsea-file-browser__tabs">
        <WorkspaceFileTabs
          projectId={projectId}
          tabs={tabs}
          activePath={activePath}
          onClose={closeFile}
          onFocus={setActivePath}
          onReorder={(ids) => setTabs((current) => reorderWorkspaceFileTabs(current, ids))}
        />
      </section>
    </div>
  );
}
```
