# 项目会话 Planner 与智能体头像栏 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 让 Session Workspace 使用系统内置 `planner` 的完整 runtime profile，并在项目菜单栏右侧显示当前项目用到的智能体头像与设置入口。

**Architecture:** 后端新增项目级 Session Planner runtime resolver，settings 保存项目覆盖 backend，Session run 创建时记录解析出的 provider 与 runtime profile 快照。前端通过项目设置 UI 管理 planner backend，通过独立项目 used agents API 渲染项目菜单栏智能体头像组。

**Tech Stack:** Node.js、Express、better-sqlite3、TypeScript、React 18、React Query、Vite、node:test。

```ts
import { agentRepo } from './repos/agents.js';
import { projectRepo } from './repos/projects.js';
import { settingsRepo } from './repos/settings.js';
import type {
  AcpBackend,
  AcpPermissionMode,
  Agent,
  AgentMemoryScope,
  AgentRuntimeBackend,
  AgentToolPolicy,
  AgentWorkspacePolicy,
} from './types.js';

export interface SessionPlannerRuntime {
  agent: Agent;
  agentId: 'planner';
  backend: AcpBackend;
  backendSource: 'project' | 'builtin';
  projectOverrideBackend: AcpBackend | null;
  permissionMode: AcpPermissionMode;
  runtimeBackend: AgentRuntimeBackend;
  toolPolicy: AgentToolPolicy;
  workspacePolicy: AgentWorkspacePolicy;
  memoryScope: AgentMemoryScope;
}

export function resolveSessionPlannerRuntime(projectId: string): SessionPlannerRuntime {
  const project = projectRepo.get(projectId);
  if (!project) throw new Error('project not found');

  const planner = agentRepo.getByBuiltinKey('planner') ?? agentRepo.getByAgentId('planner');
  if (!planner) throw new Error('system planner agent not found');
  if (!planner.default_acp_backend) throw new Error('system planner ACP backend is not configured');

  const resolution = settingsRepo.resolveForProject(project.id);
  const projectOverrideBackend = resolution?.effective.session_planner_acp_backend ?? null;
  const backend = projectOverrideBackend ?? planner.default_acp_backend;

  return {
    agent: planner,
    agentId: 'planner',
    backend,
    backendSource: projectOverrideBackend ? 'project' : 'builtin',
    projectOverrideBackend,
    permissionMode: planner.default_acp_permission_mode,
    runtimeBackend: planner.default_runtime_backend,
    toolPolicy: planner.default_tool_policy,
    workspacePolicy: planner.default_workspace_policy,
    memoryScope: planner.default_memory_scope,
  };
}

export function buildSessionPlannerRuntimeSnapshot(runtime: SessionPlannerRuntime): string {
  return JSON.stringify({
    agent_id: runtime.agentId,
    agent_global_id: runtime.agent.id,
    backend: runtime.backend,
    backend_source: runtime.backendSource,
    project_override_backend: runtime.projectOverrideBackend,
    permission_mode: runtime.permissionMode,
    runtime_backend: runtime.runtimeBackend,
    tool_policy: runtime.toolPolicy,
    workspace_policy: runtime.workspacePolicy,
    memory_scope: runtime.memoryScope,
  });
}
```

- [ ] **Step 4: 创建聚合实现**

创建 `packages/backend/src/project-used-agents.ts`：

```ts
import { roomRepo, roomAgentRepo } from './repos/rooms.js';
import { projectRepo } from './repos/projects.js';
import { resolveSessionPlannerRuntime } from './session-planner-runtime.js';
import type { ProjectUsedAgentsPayload, ProjectUsedRoomAgent } from './types.js';

export function buildProjectUsedAgents(projectId: string): ProjectUsedAgentsPayload {
  const project = projectRepo.get(projectId);
  if (!project) throw new Error('project not found');

  const plannerRuntime = resolveSessionPlannerRuntime(project.id);
  const byKey = new Map<string, ProjectUsedRoomAgent>();

  for (const room of roomRepo.listByProject(project.id)) {
    for (const agent of roomAgentRepo.listByRoom(room.id)) {
      const key = agent.global_agent_id ? `global:${agent.global_agent_id}` : `agent:${agent.agent_id}`;
      const existing = byKey.get(key);
      const binding = {
        room_id: room.id,
        room_name: room.name,
        room_agent_id: agent.id,
        acp_backend: agent.acp_backend,
        workflow_role: agent.workflow_role,
      };
      if (existing) {
        existing.room_bindings.push(binding);
        existing.acp_enabled = existing.acp_enabled || agent.acp_enabled === 1;
        if (!existing.acp_backend && agent.acp_backend) existing.acp_backend = agent.acp_backend;
        continue;
      }
      byKey.set(key, {
        kind: 'room_agent',
        global_agent_id: agent.global_agent_id,
        agent_id: agent.agent_id,
        name: agent.agent_name,
        acp_enabled: agent.acp_enabled === 1,
        acp_backend: agent.acp_backend,
        room_bindings: [binding],
      });
    }
  }

  return {
    planner: {
      kind: 'session_planner',
      agent_id: 'planner',
      name: plannerRuntime.agent.name,
      effective_acp_backend: plannerRuntime.backend,
      project_override_acp_backend: plannerRuntime.projectOverrideBackend,
      backend_source: plannerRuntime.backendSource,
      runtime_profile: {
        permission_mode: plannerRuntime.permissionMode,
        runtime_backend: plannerRuntime.runtimeBackend,
        tool_policy: plannerRuntime.toolPolicy,
        workspace_policy: plannerRuntime.workspacePolicy,
        memory_scope: plannerRuntime.memoryScope,
      },
    },
    agents: Array.from(byKey.values()).sort((left, right) => left.name.localeCompare(right.name, 'zh-CN')),
  };
}
```

```tsx
import { Bot, ChevronDown } from 'lucide-react';
import React, { useState } from 'react';
import type { ProjectUsedAgentsPayload, ProjectUsedRoomAgent, ProjectUsedRoomAgentBinding } from '../lib/types';

export function ProjectAgentStrip({
  projectId,
  data,
  onOpenPlannerSettings,
  onOpenRoomAgentSettings,
}: {
  projectId: string;
  data: ProjectUsedAgentsPayload | null | undefined;
  onOpenPlannerSettings: () => void;
  onOpenRoomAgentSettings: (binding: ProjectUsedRoomAgentBinding) => void;
}): JSX.Element {
  const [expandedAgentId, setExpandedAgentId] = useState<string | null>(null);
  const agents = data?.agents ?? [];
  const visible = agents.slice(0, 4);
  const hiddenCount = Math.max(0, agents.length - visible.length);

  return (
    <div className="deepsea-project-agents" aria-label="项目智能体">
      <button
        type="button"
        className="deepsea-project-agent-avatar"
        data-active="true"
        title={`规划师 · ${backendLabel(data?.planner.effective_acp_backend)}`}
        onClick={onOpenPlannerSettings}
      >
        <span>{initial(data?.planner.name ?? '规划师')}</span>
        <i />
      </button>
      {visible.map((agent) => (
        <div className="deepsea-project-agent-wrap" key={agent.global_agent_id ?? agent.agent_id}>
          <button
            type="button"
            className="deepsea-project-agent-avatar"
            data-active={agent.acp_enabled ? 'true' : undefined}
            title={`${agent.name} · ${backendLabel(agent.acp_backend)} · ${agent.room_bindings.length} 个房间`}
            onClick={() => {
              if (agent.room_bindings.length === 1) {
                onOpenRoomAgentSettings(agent.room_bindings[0]!);
                return;
              }
              setExpandedAgentId((current) => current === agent.agent_id ? null : agent.agent_id);
            }}
          >
            <span>{initial(agent.name)}</span>
            {agent.acp_enabled && <i />}
          </button>
          {expandedAgentId === agent.agent_id && (
            <AgentBindingMenu agent={agent} onOpenRoomAgentSettings={onOpenRoomAgentSettings} />
          )}
        </div>
      ))}
      {hiddenCount > 0 && (
        <button type="button" className="deepsea-project-agent-more" title={`${hiddenCount} 个更多智能体`}>
          +{hiddenCount}
          <ChevronDown aria-hidden="true" />
        </button>
      )}
      <span className="sr-only">{projectId}</span>
    </div>
  );
}

function AgentBindingMenu({
  agent,
  onOpenRoomAgentSettings,
}: {
  agent: ProjectUsedRoomAgent;
  onOpenRoomAgentSettings: (binding: ProjectUsedRoomAgentBinding) => void;
}): JSX.Element {
  return (
    <div className="deepsea-project-agent-menu" role="menu" aria-label={`${agent.name} 设置入口`}>
      {agent.room_bindings.map((binding) => (
        <button type="button" key={binding.room_agent_id} onClick={() => onOpenRoomAgentSettings(binding)}>
          <Bot aria-hidden="true" />
          <span>
            <strong>{binding.room_name}</strong>
            <em>{backendLabel(binding.acp_backend)} · {binding.workflow_role ?? '未设置角色'}</em>
          </span>
        </button>
      ))}
    </div>
  );
}

function initial(name: string): string {
  return name.trim().slice(0, 1).toUpperCase() || '智';
}

function backendLabel(backend: string | null | undefined): string {
  if (backend === 'codex') return 'Codex';
  if (backend === 'claudecode') return 'Claude Code';
  if (backend === 'opencode') return 'OpenCode';
  return '未启用';
}
```
