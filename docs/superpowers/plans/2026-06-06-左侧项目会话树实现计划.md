- [ ] **Step 2: 新增数据构造函数**

新增类型与函数：

```tsx
type ProjectSessionTreeProject = {
  id: string;
  name: string;
  path: string;
  active: boolean;
  sessions: ActiveSessionSummary[];
};

function buildProjectSessionTree(input: {
  projects: SessionWorkspacePayload['projectSwitcher']['projects'];
  sessions: ActiveSessionSummary[];
  currentSession: Session;
  currentProjectId: string;
  currentProjectName: string;
}): ProjectSessionTreeProject[] {
  const activeSessions = ensureCurrentActiveSessionSummary(
    input.sessions,
    input.currentSession,
    input.currentProjectId,
    input.currentProjectName,
  ).filter((session) => session.status !== 'archived');
  const projects = input.projects.map((project) => ({
    id: project.id,
    name: project.name,
    path: project.path,
    active: project.active,
    sessions: activeSessions.filter((session) => session.project_id === project.id),
  }));
  const knownProjectIds = new Set(projects.map((project) => project.id));
  const orphanSessions = activeSessions.filter((session) => !knownProjectIds.has(session.project_id));
  if (orphanSessions.length === 0) return projects;
  return [
    ...projects,
    {
      id: '__unknown__',
      name: '其他项目',
      path: '',
      active: false,
      sessions: orphanSessions,
    },
  ];
}
```
