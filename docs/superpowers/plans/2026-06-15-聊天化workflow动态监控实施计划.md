# 聊天化 Workflow 动态监控 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [x]`) syntax for tracking.

**Goal:** 移除中心聊天区顶部 `Workflow Mission` 独立面板，把 workflow 主流程和连续 workflow 状态消息合并为聊天流中的动态监控消息组，同时保留右侧状态栏与 agent run 输出能力。

**Architecture:** 在 `SessionShellView.tsx` 的 timeline 构建阶段新增 `workflow-group` item，将 workflow controller/gates/artifacts/assignments 与连续 workflow 类 transcript messages 合并为单条聊天消息。渲染层新增 `WorkflowChatMessage`，复用现有 artifact action、状态计算和消息工具栏；`RunFlowCapsule` 保持 run 消息入口，但改为更紧凑的消息内动态监控结构。

**Tech Stack:** React 18、TypeScript、Vite、Node test runner、现有 `SessionShell.test.tsx` SSR 渲染测试、CSS in `session-os.css`。

```ts
function isWorkflowTranscriptMessage(message: SessionMessage): boolean {
  const sender = `${message.sender_id} ${message.sender_name ?? ''}`.toLowerCase();
  if (sender.includes('workflow') || sender.includes('planner') || sender.includes('controller') || sender.includes('工作流')) {
```

```ts
function buildWorkflowChatGroup(
  detail: SessionDetail,
  messages: SessionMessage[],
): WorkflowChatGroup | null {
  const controller = detail.workflowController ?? null;
  const artifacts = detail.workflowArtifacts ?? [];
  const gates = detail.workflowGates ?? [];
  const assignments = detail.workflowAgentAssignments ?? [];
  if (!controller && artifacts.length === 0 && gates.length === 0 && assignments.length === 0 && messages.length === 0) {
    return null;
  }
  const timestamps = [
    ...messages.map((message) => message.created_at),
    ...artifacts.map((artifact) => artifact.created_at),
  ];
  const fallbackTime = detail.session.updated_at;
  return {
    id: controller?.workflow_run_id ?? messages[0]?.id ?? artifacts[0]?.workflow_run_id ?? 'workflow-current',
    createdAt: Math.min(...(timestamps.length ? timestamps : [fallbackTime])),
    updatedAt: Math.max(...(timestamps.length ? timestamps : [fallbackTime])),
    messages,
    controller,
    artifacts,
    gates,
    assignments,
  };
}
```

- [x] **Step 4: Add workflow helper functions**

Add:

```ts
type WorkflowChatStatus = 'pending' | 'active' | 'blocked' | 'done';

function getWorkflowChatStatus(group: WorkflowChatGroup): WorkflowChatStatus {
  if (group.controller?.blocker) return 'blocked';
  if (group.gates.some((gate) => gate.status === 'pending')) return 'blocked';
  if (group.controller) return 'active';
  if (group.messages.some((message) => /失败|异常|blocked|requires/i.test(message.content))) return 'blocked';
  if (group.messages.some((message) => /完成|已确认|done|passed/i.test(message.content))) return 'done';
  return 'pending';
}

function formatWorkflowChatStatus(status: WorkflowChatStatus): string {
  if (status === 'blocked') return '等待确认';
  if (status === 'active') return '执行中';
  if (status === 'done') return '已完成';
  return '待处理';
}

function formatWorkflowChatTitle(group: WorkflowChatGroup): string {
  const stage = group.controller?.active_stage ?? 'workflow';
  return `Workflow · ${formatWorkflowStageLabel(stage as (typeof WORKFLOW_STAGE_ORDER)[number])}`;
}

function formatWorkflowChatSummary(group: WorkflowChatGroup, sessionTitle: string): string {
  const blocker = group.controller?.blocker?.trim();
  if (blocker) return blocker;
  const nextAction = group.controller?.next_action?.trim();
  if (nextAction) return nextAction;
  const latest = [...group.messages].reverse().find((message) => message.content.trim());
  if (latest) return latest.content.trim().slice(0, 180);
  return `正在推进「${sessionTitle}」的 workflow。`;
}

function formatWorkflowChatTimeRange(group: WorkflowChatGroup): string {
  if (group.createdAt === group.updatedAt) return formatClock(group.createdAt);
  return `${formatClock(group.createdAt)} - ${formatClock(group.updatedAt)}`;
}
```
