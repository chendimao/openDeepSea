# 通用项目终端与 Skills 安全安装 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Tech Stack:** TypeScript, Express, ws, node-pty, React 18, Vite, @xterm/xterm, @xterm/addon-fit, TanStack Query, node:test.

```ts
export type RestrictedSkillsCommand =
  | { ok: true; command: string | null; args: string[]; builtin: 'clear' | 'exit' | 'pwd' | null }
  | { ok: false; reason: string };

const BLOCKED_PATTERNS = [
  /;/,
  /&&/,
  /\|\|/,
  /\|/,
  />>?/,
  /</,
  /`/,
  /\$\(/,
  /^[A-Za-z_][A-Za-z0-9_]*=/,
];

const ALLOWED_SKILLS_ACTIONS = new Set(['find', 'add', 'check', 'update']);

export function parseRestrictedSkillsCommand(input: string): RestrictedSkillsCommand {
  const trimmed = input.trim();
  if (!trimmed) return { ok: false, reason: '请输入 skills 安装相关命令。' };
  if (trimmed === 'clear') return { ok: true, command: null, args: [], builtin: 'clear' };
  if (trimmed === 'exit') return { ok: true, command: null, args: [], builtin: 'exit' };
  if (trimmed === 'pwd') return { ok: true, command: null, args: [], builtin: 'pwd' };

  for (const pattern of BLOCKED_PATTERNS) {
    if (pattern.test(trimmed)) {
      return { ok: false, reason: 'Skills 安装终端不允许 shell 连接符、重定向、命令替换或环境变量前缀。' };
    }
  }

  const parts = splitCommand(trimmed);
  if (parts.length === 0) return { ok: false, reason: '请输入 skills 安装相关命令。' };

  if (parts[0] === 'skills') return parseSkills(parts);
  if (parts[0] === 'npx') return parseNpx(parts);
  return { ok: false, reason: '只允许执行 npx skills 命令或 skills 命令。' };
}

function parseSkills(parts: string[]): RestrictedSkillsCommand {
  const action = parts[1];
  if (!action || !ALLOWED_SKILLS_ACTIONS.has(action)) {
    return { ok: false, reason: '只允许 skills find/add/check/update。' };
  }
  return { ok: true, command: 'skills', args: parts.slice(1), builtin: null };
}

function parseNpx(parts: string[]): RestrictedSkillsCommand {
  const args = parts.slice(1);
  const skillsIndex = args[0] === '--yes' ? 1 : 0;
  if (args[skillsIndex] !== 'skills') {
    return { ok: false, reason: '只允许 npx skills 命令或 npx --yes skills 命令。' };
  }
  const action = args[skillsIndex + 1];
  if (!action || !ALLOWED_SKILLS_ACTIONS.has(action)) {
    return { ok: false, reason: '只允许 npx skills find/add/check/update。' };
  }
  return { ok: true, command: 'npx', args, builtin: null };
}

function splitCommand(input: string): string[] {
  const parts: string[] = [];
  let current = '';
  let quote: '"' | "'" | null = null;
  for (const char of input) {
    if ((char === '"' || char === "'") && quote === null) {
      quote = char;
      continue;
    }
    if (quote === char) {
      quote = null;
      continue;
    }
    if (!quote && /\s/.test(char)) {
      if (current) parts.push(current);
      current = '';
      continue;
    }
    current += char;
  }
  if (quote) return [];
  if (current) parts.push(current);
  return parts;
}
```

```ts
import { EventEmitter } from 'node:events';
import { cwd as processCwd, env, platform } from 'node:process';
import { nanoid } from 'nanoid';
import { spawn, type IPty } from 'node-pty';
import { projectRepo } from '../repos/projects.js';
import { parseRestrictedSkillsCommand } from './restricted-skills-shell.js';
import type { CreateTerminalSessionInput, TerminalProfile, TerminalServerEvent, TerminalSessionInfo, TerminalStatus } from './types.js';

type ManagedSession = TerminalSessionInfo & {
  pty: IPty | null;
  emitter: EventEmitter;
  bufferedOutput: string[];
  profile: TerminalProfile;
  pendingRestrictedCommand: boolean;
};

const sessions = new Map<string, ManagedSession>();

export const terminalService = {
  create(input: CreateTerminalSessionInput): TerminalSessionInfo {
    const cwd = resolveCwd(input.projectId);
    const session = createSession(input.profile, cwd, input.cols, input.rows);
    sessions.set(session.id, session);
    return toInfo(session);
  },

  get(id: string): TerminalSessionInfo | null {
    const session = sessions.get(id);
    return session ? toInfo(session) : null;
  },

  subscribe(id: string, listener: (event: TerminalServerEvent) => void): (() => void) | null {
    const session = sessions.get(id);
    if (!session) return null;
    const wrapped = (event: TerminalServerEvent) => listener(event);
    session.emitter.on('event', wrapped);
    listener({ type: 'terminal:ready', sessionId: session.id, cwd: session.cwd, profile: session.profile });
    for (const data of session.bufferedOutput) listener({ type: 'terminal:output', sessionId: session.id, data });
    return () => session.emitter.off('event', wrapped);
  },

  input(id: string, data: string): boolean {
    const session = sessions.get(id);
    if (!session || session.status !== 'running') return false;
    if (session.profile === 'project_shell') {
      session.pty?.write(data);
      return true;
    }
    handleRestrictedInput(session, data);
    return true;
  },

  resize(id: string, cols: number, rows: number): boolean {
    const session = sessions.get(id);
    if (!session || session.status !== 'running') return false;
    session.pty?.resize(cols, rows);
    return true;
  },

  kill(id: string, status: TerminalStatus = 'killed'): boolean {
    const session = sessions.get(id);
    if (!session) return false;
    session.status = status;
    session.endedAt = Date.now();
    session.pty?.kill();
    emit(session, { type: 'terminal:status', sessionId: id, status });
    return true;
  },
};

function createSession(profile: TerminalProfile, cwd: string, cols: number, rows: number): ManagedSession {
  const id = nanoid();
  const session: ManagedSession = {
    id,
    profile,
    cwd,
    status: 'running',
    startedAt: Date.now(),
    endedAt: null,
    exitCode: null,
    signal: null,
    pty: null,
    emitter: new EventEmitter(),
    bufferedOutput: [],
    pendingRestrictedCommand: false,
  };
  if (profile === 'project_shell') {
    session.pty = spawn(resolveShell(), [], { name: 'xterm-256color', cols, rows, cwd, env: buildEnv() });
    wirePty(session);
  } else {
    writeOutput(session, [
      'Skills 安装终端，只允许执行 skills 安装相关命令。\\r\\n',
      '可用示例：npx skills find react / npx skills add <package> / skills check / skills update\\r\\n',
      `${cwd} skills> `,
    ].join(''));
  }
  return session;
}

function handleRestrictedInput(session: ManagedSession, data: string): void {
  if (session.pty) {
    session.pty.write(data);
    return;
  }
  if (data === '\\r' || data === '\\n') return;
  const line = data.replace(/\\r?\\n$/, '');
  writeOutput(session, `${line}\\r\\n`);
  const parsed = parseRestrictedSkillsCommand(line);
  if (!parsed.ok) {
    writeOutput(session, `${parsed.reason}\\r\\n${session.cwd} skills> `);
    return;
  }
  if (parsed.builtin === 'clear') {
    writeOutput(session, '\\x1bc');
    writeOutput(session, `${session.cwd} skills> `);
    return;
  }
  if (parsed.builtin === 'pwd') {
    writeOutput(session, `${session.cwd}\\r\\n${session.cwd} skills> `);
    return;
  }
  if (parsed.builtin === 'exit') {
    finish(session, 0, null);
    return;
  }
  if (!parsed.command) return;
  session.pty = spawn(parsed.command, parsed.args, { name: 'xterm-256color', cols: 100, rows: 30, cwd: session.cwd, env: buildEnv() });
  wirePty(session);
}

function wirePty(session: ManagedSession): void {
  session.pty?.onData((data) => writeOutput(session, data));
  session.pty?.onExit(({ exitCode, signal }) => {
    const profile = session.profile;
    session.pty = null;
    if (profile === 'skills_install' && session.status === 'running') {
      writeOutput(session, `\\r\\n${session.cwd} skills> `);
      emit(session, { type: 'platform_skills:refresh_requested' });
      return;
    }
    finish(session, exitCode, signal ?? null);
  });
}

function finish(session: ManagedSession, exitCode: number | null, signal: string | null): void {
  session.status = session.status === 'killed' ? 'killed' : 'exited';
  session.exitCode = exitCode;
  session.signal = signal;
  session.endedAt = Date.now();
  emit(session, { type: 'terminal:exit', sessionId: session.id, exitCode, signal });
}

function writeOutput(session: ManagedSession, data: string): void {
  session.bufferedOutput.push(data);
  if (session.bufferedOutput.length > 200) session.bufferedOutput.shift();
  emit(session, { type: 'terminal:output', sessionId: session.id, data });
}

function emit(session: ManagedSession, event: TerminalServerEvent): void {
  session.emitter.emit('event', event);
}

function toInfo(session: ManagedSession): TerminalSessionInfo {
  return {
    id: session.id,
    profile: session.profile,
    cwd: session.cwd,
    status: session.status,
    startedAt: session.startedAt,
    endedAt: session.endedAt,
    exitCode: session.exitCode,
    signal: session.signal,
  };
}

function resolveCwd(projectId: string | null | undefined): string {
  if (projectId) {
    const project = projectRepo.get(projectId);
    if (!project) throw new Error('project not found');
    return project.path;
  }
  return processCwd();
}

function resolveShell(): string {
  if (env.SHELL?.trim()) return env.SHELL;
  if (platform === 'win32') return 'powershell.exe';
  return 'zsh';
}

function buildEnv(): NodeJS.ProcessEnv {
  const allowed = ['HOME', 'PATH', 'SHELL', 'TERM', 'LANG', 'LC_ALL', 'USER', 'TMPDIR'];
  const next: NodeJS.ProcessEnv = {};
  for (const key of allowed) {
    if (env[key]) next[key] = env[key];
  }
  next.TERM = 'xterm-256color';
  return next;
}
```

```ts
import type { WebSocket } from 'ws';
import type { WsClientEvent } from '../types.js';
import { terminalService } from './service.js';

const subscriptions = new WeakMap<WebSocket, Map<string, () => void>>();

export function handleTerminalSocketEvent(socket: WebSocket, event: WsClientEvent): boolean {
  if (!event.type.startsWith('terminal:')) return false;

  if (event.type === 'terminal:subscribe') {
    const cleanup = terminalService.subscribe(event.sessionId, (serverEvent) => {
      if (socket.readyState === socket.OPEN) socket.send(JSON.stringify(serverEvent));
    });
    if (cleanup) {
      const socketSubscriptions = subscriptions.get(socket) ?? new Map<string, () => void>();
      socketSubscriptions.set(event.sessionId, cleanup);
      subscriptions.set(socket, socketSubscriptions);
    }
    return true;
  }

  if (event.type === 'terminal:unsubscribe') {
    unsubscribe(socket, event.sessionId);
    return true;
  }

  if (event.type === 'terminal:input') {
    terminalService.input(event.sessionId, event.data);
    return true;
  }

  if (event.type === 'terminal:resize') {
    terminalService.resize(event.sessionId, event.cols, event.rows);
    return true;
  }

  if (event.type === 'terminal:kill') {
    terminalService.kill(event.sessionId);
    return true;
  }

  return true;
}

export function removeTerminalSocket(socket: WebSocket): void {
  const socketSubscriptions = subscriptions.get(socket);
  if (!socketSubscriptions) return;
  for (const cleanup of socketSubscriptions.values()) cleanup();
  subscriptions.delete(socket);
}

function unsubscribe(socket: WebSocket, sessionId: string): void {
  const socketSubscriptions = subscriptions.get(socket);
  const cleanup = socketSubscriptions?.get(sessionId);
  cleanup?.();
  socketSubscriptions?.delete(sessionId);
}
```

```tsx
import '@xterm/xterm/css/xterm.css';
import { FitAddon } from '@xterm/addon-fit';
import { Terminal } from '@xterm/xterm';
import { Maximize2, Square, X } from 'lucide-react';
import { useEffect, useRef, useState } from 'react';
import { api } from '../lib/api';
import type { TerminalProfile, TerminalSessionInfo, TerminalStatus } from '../lib/types';
import { roomSocket, type WsServerEvent } from '../lib/ws';
import { cn } from '../lib/utils';

type TerminalPanelProps = {
  profile: TerminalProfile;
  projectId?: string | null;
  title: string;
  className?: string;
  onClose?: () => void;
  onRefreshRequested?: () => void;
};

export function TerminalPanel({
  profile,
  projectId = null,
  title,
  className,
  onClose,
  onRefreshRequested,
}: TerminalPanelProps): JSX.Element {
  const hostRef = useRef<HTMLDivElement | null>(null);
  const terminalRef = useRef<Terminal | null>(null);
  const fitRef = useRef<FitAddon | null>(null);
  const sessionRef = useRef<TerminalSessionInfo | null>(null);
  const [session, setSession] = useState<TerminalSessionInfo | null>(null);
  const [status, setStatus] = useState<TerminalStatus | 'connecting'>('connecting');

  useEffect(() => {
    const terminal = new Terminal({
      cursorBlink: true,
      convertEol: true,
      fontFamily: 'var(--font-mono, ui-monospace, SFMono-Regular, Menlo, monospace)',
      fontSize: 12,
      theme: {
        background: '#07090f',
        foreground: '#d6dae6',
        cursor: '#ffffff',
      },
    });
    const fit = new FitAddon();
    terminal.loadAddon(fit);
    terminalRef.current = terminal;
    fitRef.current = fit;
    if (hostRef.current) {
      terminal.open(hostRef.current);
      fit.fit();
    }

    let disposed = false;
    const cols = terminal.cols || 100;
    const rows = terminal.rows || 30;
    void api.createTerminalSession({ profile, projectId, cols, rows }).then((created) => {
      if (disposed) return;
      setSession(created);
      setStatus(created.status);
      roomSocket.subscribeTerminal(created.id);
      terminal.writeln(`Connected: ${created.cwd}`);
    }).catch((err) => {
      if (disposed) return;
      setStatus('failed');
      terminal.writeln(`Terminal failed: ${(err as Error).message}`);
    });

    const disposeData = terminal.onData((data) => {
      const id = sessionRef.current?.id;
      if (id) roomSocket.sendTerminalInput(id, data);
    });
    const disposeResize = terminal.onResize(({ cols: nextCols, rows: nextRows }) => {
      const id = sessionRef.current?.id;
      if (id) roomSocket.resizeTerminal(id, nextCols, nextRows);
    });
    const onWindowResize = () => fit.fit();
    window.addEventListener('resize', onWindowResize);

    return () => {
      disposed = true;
      const id = sessionRef.current?.id;
      if (id) roomSocket.unsubscribeTerminal(id);
      window.removeEventListener('resize', onWindowResize);
      disposeData.dispose();
      disposeResize.dispose();
      terminal.dispose();
    };
  }, [profile, projectId]);

  useEffect(() => {
    sessionRef.current = session;
  }, [session]);

  useEffect(() => {
    return roomSocket.on((event: WsServerEvent) => {
      const active = sessionRef.current;
      if (!active) return;
      if (event.type === 'terminal:output' && event.sessionId === active.id) {
        terminalRef.current?.write(event.data);
      }
      if (event.type === 'terminal:status' && event.sessionId === active.id) {
        setStatus(event.status);
      }
      if (event.type === 'terminal:exit' && event.sessionId === active.id) {
        setStatus('exited');
        onRefreshRequested?.();
      }
      if (event.type === 'platform_skills:refresh_requested') {
        onRefreshRequested?.();
      }
    });
  }, [onRefreshRequested]);

  const kill = async () => {
    if (!session) return;
    roomSocket.killTerminal(session.id);
    await api.killTerminalSession(session.id).catch(() => undefined);
    setStatus('killed');
    onRefreshRequested?.();
  };

  return (
    <section className={cn('terminal-panel', className)}>
      <header className="terminal-panel__header">
        <div>
          <b>{title}</b>
          <span>{session?.cwd ?? '连接中'} · {status}</span>
        </div>
        <div>
          <button type="button" aria-label="适配终端尺寸" onClick={() => fitRef.current?.fit()}><Maximize2 /></button>
          <button type="button" aria-label="终止终端" onClick={kill}><Square /></button>
          {onClose ? <button type="button" aria-label="关闭终端" onClick={onClose}><X /></button> : null}
        </div>
      </header>
      <div ref={hostRef} className="terminal-panel__body" />
    </section>
  );
}
```
