# 任务风险门禁与多智能体调度闭环 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 实现“轻量任务自动执行，中高风险任务先确认”的多智能体调度闭环 MVP，并让 Planner、Scheduler、Agent event、Review/Verify 返工链路有结构化状态可追踪。

**Architecture:** 复用现有 `task-profile`、`supervisor`、`graph/state`、`graph/nodes`、`workflow_steps`、`agent_runs` 和 task events。新增风险评估、approval card、结构化 agent event 与可测试的调度守卫；第一阶段不做数据库迁移，优先把结构化数据保存在 `workflow_runs.graph_state`、artifact metadata 和 task event metadata。

**Tech Stack:** Node.js 20、TypeScript、zod、node:test、Express、SQLite、React 18、Vite、现有 ACP adapter 与 LangGraph workflow runtime。

---

## Scope Check

本计划覆盖一个可独立验收的 MVP：

- 后端可判断任务风险并决定是否自动通过 approval gate。
- Planner 计划和 workflow graph state 可携带 `riskAssessment` 与 `approvalCard`。
- 中高风险任务创建结构化确认 artifact/event，用户确认前不创建执行型 agent run。
- 子智能体事件可通过统一协议记录到 task timeline。
- 并行调度的写入冲突和高风险范围规则被单元测试覆盖。
- 前端类型和会话时间线能展示风险门禁与 agent event 摘要。

不在本计划内实现完全动态临时 workflow graph、数据库表拆分、复杂可视化执行树。

```ts
import { inferTaskProfile } from './task-profile.js';
import type { ParsedVerificationCommand } from './plan-parser.js';

export type TaskKind =
  | 'chat_answer'
  | 'brainstorming'
  | 'code_review'
  | 'bug_fix'
  | 'frontend_change'
  | 'backend_change'
  | 'fullstack_change'
  | 'test_only'
  | 'docs_only'
  | 'ops_or_config'
  | 'unknown';

export type TaskRiskLevel = 'low' | 'medium' | 'high';
export type WorkflowExecutionMode = 'serial' | 'parallel' | 'hybrid';

export interface TaskRiskAssessmentInput {
  title: string;
  description: string | null | undefined;
  scopeRead: string[];
  scopeWrite: string[];
  acceptance?: string[];
  verificationCommands?: ParsedVerificationCommand[];
}

export interface TaskRiskAssessment {
  taskKind: TaskKind;
  riskLevel: TaskRiskLevel;
  requiresApproval: boolean;
  approvalReason: string;
  confidence: number;
  reasons: string[];
  scopeRead: string[];
  scopeWrite: string[];
  verificationCommands: ParsedVerificationCommand[];
}

export interface ApprovalCard {
  riskLevel: Exclude<TaskRiskLevel, 'low'>;
  taskKind: TaskKind;
  summary: string;
  approvalReason: string;
  agents: string[];
  executionMode: WorkflowExecutionMode;
  scopeRead: string[];
  scopeWrite: string[];
  verification: ParsedVerificationCommand[];
  risks: string[];
  assumptions: string[];
}

export function assessTaskRisk(input: TaskRiskAssessmentInput): TaskRiskAssessment {
  const profile = inferTaskProfile({
    title: input.title,
    description: input.description ?? '',
    scopeRead: input.scopeRead,
    scopeWrite: input.scopeWrite,
    acceptance: input.acceptance,
  });
  const text = [
    input.title,
    input.description ?? '',
    ...input.scopeRead,
    ...input.scopeWrite,
    ...(input.acceptance ?? []),
  ].join('\n').toLowerCase();
  const reasons = [...profile.reasons];
  const highSignals = collectHighRiskSignals(input.scopeWrite, text);
  const mediumSignals = collectMediumRiskSignals(input.scopeWrite, text);
  const taskKind = mapTaskKind(profile.taskType, text);

  let riskLevel: TaskRiskLevel = 'low';
  let approvalReason = 'Low risk small or documentation-only task.';
  if (highSignals.length > 0) {
    riskLevel = 'high';
    approvalReason = `High risk because ${highSignals.join(', ')}.`;
    reasons.push(...highSignals);
  } else if (mediumSignals.length > 0 || profile.domains.length > 1) {
    riskLevel = 'medium';
    approvalReason = mediumSignals.length > 0
      ? `Medium risk because ${mediumSignals.join(', ')}.`
      : 'Medium risk because frontend and backend domains are both involved.';
    reasons.push(...mediumSignals);
  }

  return {
    taskKind,
    riskLevel,
    requiresApproval: riskLevel !== 'low' || profile.confidence < 0.45,
    approvalReason: profile.confidence < 0.45 && riskLevel === 'low'
      ? 'Classifier confidence is low; ask for confirmation before execution.'
      : approvalReason,
    confidence: profile.confidence,
    reasons,
    scopeRead: input.scopeRead,
    scopeWrite: input.scopeWrite,
    verificationCommands: input.verificationCommands ?? [],
  };
}

export function buildApprovalCard(input: {
  assessment: TaskRiskAssessment;
  agents: string[];
  executionMode: WorkflowExecutionMode;
  risks?: string[];
  assumptions?: string[];
}): ApprovalCard {
  if (input.assessment.riskLevel === 'low') {
    throw new Error('approval card requires medium or high risk assessment');
  }
  return {
    riskLevel: input.assessment.riskLevel,
    taskKind: input.assessment.taskKind,
    summary: input.assessment.approvalReason,
    approvalReason: input.assessment.approvalReason,
    agents: input.agents,
    executionMode: input.executionMode,
    scopeRead: input.assessment.scopeRead,
    scopeWrite: input.assessment.scopeWrite,
    verification: input.assessment.verificationCommands,
    risks: input.risks ?? input.assessment.reasons,
    assumptions: input.assumptions ?? [],
  };
}

function mapTaskKind(taskType: string, text: string): TaskKind {
  if (/(review|审查|评审)/i.test(text)) return 'code_review';
  if (/(test|测试|回归)/i.test(text) && !/(实现|开发|新增|修改|fix|修复)/i.test(text)) return 'test_only';
  switch (taskType) {
    case 'frontend_feature':
      return 'frontend_change';
    case 'backend_feature':
      return 'backend_change';
    case 'fullstack_feature':
      return 'fullstack_change';
    case 'bugfix':
      return 'bug_fix';
    case 'documentation':
    case 'presentation':
      return 'docs_only';
    default:
      return 'unknown';
  }
}

function collectHighRiskSignals(scopeWrite: string[], text: string): string[] {
  const signals = new Set<string>();
  for (const scope of scopeWrite.map((item) => item.toLowerCase())) {
    const file = scope.split(/[\\/]/).at(-1) ?? scope;
    if (['package.json', 'package-lock.json', 'pnpm-lock.yaml', 'yarn.lock'].includes(file)) signals.add('dependency or lockfile change');
    if (['tsconfig.json', 'vite.config.ts', 'electron-builder.json'].includes(file)) signals.add('root config change');
    if (scope.includes('migration') || scope.includes('migrations') || scope.endsWith('.sql')) signals.add('database migration');
    if (scope.includes('/db') || scope.includes('\\db')) signals.add('database code change');
  }
  if (/(delete|删除|drop table|credential|secret|api key|权限|沙箱|sandbox)/i.test(text)) signals.add('security or destructive operation');
  return [...signals];
}

function collectMediumRiskSignals(scopeWrite: string[], text: string): string[] {
  const signals = new Set<string>();
  const hasFrontend = scopeWrite.some((scope) => scope.includes('packages/frontend')) || /frontend|前端|react|ui|ux/i.test(text);
  const hasBackend = scopeWrite.some((scope) => scope.includes('packages/backend')) || /backend|后端|api|sqlite|workflow/i.test(text);
  if (hasFrontend && hasBackend) signals.add('frontend and backend coordination');
  if (/shared|contract|schema|types|workflow|状态机|编排|调度/i.test(text)) signals.add('shared contract or workflow behavior');
  if (scopeWrite.length >= 4) signals.add('multi-file change');
  return [...signals];
}
```

```ts
const structuredAgentEventSchema = z.object({

  detail: z.string().optional(),
  progress: z.number().min(0).max(100).optional(),
```

```ts
    recommendation: z.string().optional(),
    impact: z.string().min(1),
  }).optional(),
  createdAt: z.number().int().min(0),
});

export function parseStructuredAgentEvent(value: unknown): StructuredAgentEvent {
```

```ts
import { isAbsolute, relative, resolve } from 'node:path';

export function scopeWritesConflict(left: string[], right: string[], projectPath: string): boolean {
  if (left.length === 0 || right.length === 0) return false;
  const normalizedLeft = left.map((scope) => normalizeScopePath(scope, projectPath)).filter((scope): scope is string => scope !== null);
  const normalizedRight = right.map((scope) => normalizeScopePath(scope, projectPath)).filter((scope): scope is string => scope !== null);
  return normalizedLeft.some((leftScope) =>
    normalizedRight.some((rightScope) => scopePathConflicts(leftScope, rightScope)),
  );
}

export function scopeWritesRequireSerial(writes: string[], projectPath: string): boolean {
  return writes
    .map((scope) => normalizeScopePath(scope, projectPath))
    .filter((scope): scope is string => scope !== null)
    .some((scope) => scopePathRequiresSerial(scope));
}

function scopePathRequiresSerial(scope: string): boolean {
  if (isBroadScopePath(scope)) return true;
  const segments = scope.split('/').filter(Boolean);
  const basename = segments.at(-1) ?? scope;
  if ([
    'package.json',
    'package-lock.json',
    'npm-shrinkwrap.json',
    'pnpm-lock.yaml',
    'yarn.lock',
    'tsconfig.json',
    'turbo.json',
    'nx.json',
    'lerna.json',
    'rush.json',
    'pnpm-workspace.yaml',
    'pnpm-workspace.yml',
    'workspace.json',
    '.gitlab-ci.yml',
    '.gitlab-ci.yaml',
    'azure-pipelines.yml',
    'Jenkinsfile',
    'Dockerfile',
    'docker-compose.yml',
    'compose.yml',
    'compose.yaml',
  ].includes(basename)) return true;
  if (scope.includes('/migrations/') || scope.endsWith('.sql')) return true;
  if (scope.includes('/schema/') || scope.includes('/shared/') || scope.includes('/types/')) return true;
  return false;
}

function normalizeScopePath(scope: string, projectPath: string): string | null {
  const trimmed = scope.trim();
  if (!trimmed) return null;
  if (trimmed === '*' || trimmed === '.' || trimmed === '/') return trimmed;
  const absolute = isAbsolute(trimmed) ? trimmed : resolve(projectPath, trimmed);
  const relativePath = relative(projectPath, absolute);
  if (relativePath.startsWith('..')) return absolute.replace(/\\/g, '/');
  return relativePath.replace(/\\/g, '/');
}

function scopePathConflicts(left: string, right: string): boolean {
  if (left === right) return true;
  if (isBroadScopePath(left) || isBroadScopePath(right)) return true;
  return right.startsWith(`${left}/`) || left.startsWith(`${right}/`);
}

function isBroadScopePath(scope: string): boolean {
  return scope === '.' || scope === '/' || scope === '*';
}
```
